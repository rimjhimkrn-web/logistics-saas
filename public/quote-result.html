// --- PERMANENTLY LOCKED CHECKOUT SCRIPT --- async function handleSecureCheckout(shipmentId, customerEmail) { try { // 1. Retrieve active Supabase session token for institutional authorization const { data: sessionData, error: sessionError } = await supabase.auth.getSession(); if (sessionError || !sessionData?.session) { throw new Error("Authentication required. Please sign in again."); } const accessToken = sessionData.session.access_token; // 2. Request server-side secure order creation (passing shipment_id, NOT price) const orderResponse = await `${SUPABASE_PROJECT_URL}/functions/v1/create-razorpay-order`; const orderFetch = await fetch(orderResponse, { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${accessToken}` }, body: JSON.stringify({ shipment_id: shipmentId, customer_email: customerEmail }) }); if (!orderFetch.ok) { const errText = await orderFetch.text(); throw new Error(`Server order initialization failed: ${errText}`); } const orderData = await orderFetch.json(); if (!orderData.success || !orderData.order_id) { throw new Error(orderData.error || "Failed to initialize secure order token."); } // 3. Initialize Razorpay Checkout Modal const options = { key: orderData.key_id, amount: orderData.amount, // Secured from backend krim_quotes currency: orderData.currency || "INR", name: "KRIM Logistics OS", description: `Freight Settlement for Shipment #${shipmentId}`, order_id: orderData.order_id, handler: async function (res) { // 4. Send payment proof to verified server-side verification endpoint try { const verifyResponse = await fetch( `${SUPABASE_PROJECT_URL}/functions/v1/process-payment-`, { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${accessToken}` }, body: JSON.stringify({ shipment_id: shipmentId, payment_id: res.razorpay_payment_id, order_id: res.razorpay_order_id, signature: res.razorpay_signature, customer_email: customerEmail }) } ); if (!verifyResponse.ok) { throw new Error("Payment verification failed on the server."); } const verifyData = await verifyResponse.json(); if (!verifyData.success) { throw new Error(verifyData.error || "Cryptographic signature validation failed."); } // 5. Display Verified Success Screen showVerifiedSuccessScreen(verifyData); } catch (verifyErr) { alert(`Payment verification error: ${verifyErr.message}`); } }, prefill: { email: customerEmail }, theme: { color: "#0F172A" } }; const rzp = new Razorpay(options); rzp.open(); } catch (err) { alert(`Checkout Error: ${err.message}`); } }
